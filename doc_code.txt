function fetchLiveSafetyData(targetDateStr) {
  try {
    const safetySheet = SpreadsheetApp.openById(LIVE_SAFETY_DB_ID).getSheetByName('Safety Log');
    if (!safetySheet) return { summary: { totals: [] }, register: [] };
   
    const safetyData = safetySheet.getDataRange().getValues();
    const headers = safetyData.shift();
   
    const colType = headers.findIndex(h => h.toString().includes('Observation Type'));
    const colImpact = headers.findIndex(h => h.toString().includes('Potential Severity'));
    const colStatus = headers.findIndex(h => h.toString().includes('Status'));
    const colDate = headers.findIndex(h => h.toString().includes('Date and Time'));
    const colTargetDate = headers.findIndex(h => h.toString().includes('Target Date of Closure'));
    const colActualDate = headers.findIndex(h => h.toString().includes('Date Closed'));
    const colPrevAction = headers.findIndex(h => h.toString().includes('Preventive Action'));
    const colRemarks = headers.findIndex(h => h.toString().includes('Remarks'));
    const colDept = headers.findIndex(h => h.toString().toLowerCase().includes('department') || h.toString().toLowerCase().includes('dept'));
    let summaryTotals = {
      "Unsafe Act": { total: 0, open: 0, overdue: 0, closed: 0, high: 0, medium: 0, low: 0 },
      "Unsafe Condition": { total: 0, open: 0, overdue: 0, closed: 0, high: 0, medium: 0, low: 0 },
      "Near Miss": { total: 0, open: 0, overdue: 0, closed: 0, high: 0, medium: 0, low: 0 }
    };
   
    let totalDaysToClose = 0;
    let closedCountForAvg = 0;
    let registerArray = [];


  const tDateObj = new Date(targetDateStr);
    tDateObj.setHours(0, 0, 0, 0); // Normalize time
    const targetYear = tDateObj.getFullYear();
    const targetMonth = tDateObj.getMonth();
   
   safetyData.forEach((row, index) => {
 let originalDate = new Date(row[colDate]);
      if (isNaN(originalDate)) return;
     
      let repDate = new Date(originalDate);
      // We set the target Date Object to 23:59:59 so it includes the ENTIRE target day.
      let endOfDayTarget = new Date(tDateObj);
      endOfDayTarget.setHours(23, 59, 59, 999);


      // STRICT MTD FILTER: Must be this month, and CANNOT be after the selected date
      if (repDate.getFullYear() !== targetYear || repDate.getMonth() !== targetMonth) return;
      if (repDate > endOfDayTarget) return;


      let typeStr = row[colType] ? row[colType].toString() : "";
      let type = typeStr.includes('Unsafe Act') ? 'Unsafe Act' :
                 typeStr.includes('Unsafe Condition') ? 'Unsafe Condition' :
                 typeStr.includes('Near Miss') ? 'Near Miss' : 'Unknown';
                 
      if (type === 'Unknown') return;
     
      let impactStr = row[colImpact] ? row[colImpact].toString().toLowerCase() : "";
      let impact = impactStr.includes('high') ? 'High' :
                   impactStr.includes('medium') ? 'Medium' :
                   impactStr.includes('low') ? 'Low' : '-';
                   
      let statusStr = row[colStatus] ? row[colStatus].toString().toLowerCase() : "";
      let status = statusStr.includes('close') ? 'Closed' : 'Open';
     
      let closeDate = row[colActualDate] ? new Date(row[colActualDate]) : null;
      let daysToClose = 0;
      if (closeDate && !isNaN(closeDate) && !isNaN(repDate)) {
        daysToClose = Math.ceil((closeDate - repDate) / (1000 * 60 * 60 * 24));
        if (daysToClose < 0) daysToClose = 0;
      }
     
      summaryTotals[type].total++;
      if (status === 'Closed') {
        summaryTotals[type].closed++;
        totalDaysToClose += daysToClose;
        closedCountForAvg++;
      } else {
        summaryTotals[type].open++;
        if (row[colTargetDate]) {
          let tDate = new Date(row[colTargetDate]);
          if (!isNaN(tDate) && tDate < new Date()) summaryTotals[type].overdue++;
        }
      }
     
      if (impact === 'High') summaryTotals[type].high++;
      else if (impact === 'Medium') summaryTotals[type].medium++;
      else if (impact === 'Low') summaryTotals[type].low++;
     
    let dFormat = Utilities.formatDate(originalDate, Session.getScriptTimeZone(), "yyyy-MM-dd");
      let tFormat = Utilities.formatDate(originalDate, Session.getScriptTimeZone(), "HH:mm:ss");
      let targetFormat = row[colTargetDate] && !isNaN(new Date(row[colTargetDate])) ? Utilities.formatDate(new Date(row[colTargetDate]), Session.getScriptTimeZone(), "yyyy-MM-dd") : "-";
      let actFormat = closeDate && !isNaN(closeDate) ? Utilities.formatDate(closeDate, Session.getScriptTimeZone(), "yyyy-MM-dd") : "-";
     
registerArray.push({
        sr: registerArray.length + 1, date: dFormat, time: tFormat, type: type,
        desc: row[10] || "-", location: row[9] || "-", reporter: row[7] || "-",
dept: row[22] || "-", impact: impact, cause: row[12] || "-",
        corrective: row[13] || "-", preventive: colPrevAction > -1 ? row[colPrevAction] : "-",
        assigned: row[14] || "-", targetDate: targetFormat, actualDate: actFormat,
        status: status, daysToClose: daysToClose, remarks: colRemarks > -1 ? row[colRemarks] : "-"
      });
    });
   
    let totalsArray = [];
    let totalsRow = { type: "Total", total: 0, open: 0, overdue: 0, closed: 0, high: 0, medium: 0, low: 0 };
    ["Unsafe Act", "Unsafe Condition", "Near Miss"].forEach(t => {
      let r = summaryTotals[t]; r.type = t; totalsArray.push(r);
      totalsRow.total += r.total; totalsRow.open += r.open; totalsRow.overdue += r.overdue;
      totalsRow.closed += r.closed; totalsRow.high += r.high; totalsRow.medium += r.medium; totalsRow.low += r.low;
    });
    totalsArray.push(totalsRow);
   
    let closureRate = totalsRow.total > 0 ? ((totalsRow.closed / totalsRow.total) * 100).toFixed(2) + "%" : "0%";
    let avgDays = closedCountForAvg > 0 ? (totalDaysToClose / closedCountForAvg).toFixed(2) : "0";
   
    return {
      summary: { totals: totalsArray, closureRate: closureRate, avgDays: avgDays },
      register: registerArray
    };
  } catch (e) {
    Logger.log("Safety Fetch Error: " + e);
    return { summary: { totals: [], closureRate: "0%", avgDays: "0" }, register: [] };
  }
}


function fetchLivePTWData(targetDateStr) {
  try {
  const tDateObj = new Date(targetDateStr);
    tDateObj.setHours(0, 0, 0, 0);
   
   const ptwSheet = SpreadsheetApp.openById(LIVE_PTW_DB_ID).getSheetByName('PTW_Master_Data');
    if (!ptwSheet) return null;
   
    const ptwData = ptwSheet.getDataRange().getValues();
    const headers = ptwData.shift();
    const colDate = headers.findIndex(h => h.toString().includes('Permit Start Date'));
   
// Dynamically find columns for the Permit Details table
    const colPermitNo = headers.findIndex(h => h.toString().toLowerCase().match(/permit\s*(no|#|num)|ptw\s*(no|#|num)/i));
    const colVendor = headers.findIndex(h => h.toString().toLowerCase().match(/vendor|contractor|agency/));
    const colLocation = headers.findIndex(h => h.toString().toLowerCase().match(/location|area|plant/));
    const colDesc = headers.findIndex(h => h.toString().toLowerCase().match(/description|nature of work|details/));
    const colType = headers.findIndex(h => h.toString().toLowerCase().match(/permit type|type of permit/));
   
    const permits = {
      gen: { t: 0, m: 0 }, hot: { t: 0, m: 0 }, hgt: { t: 0, m: 0 }, cnf: { t: 0, m: 0 },
      exc: { t: 0, m: 0 }, lin: { t: 0, m: 0 }, hst: { t: 0, m: 0 },
      todayDetails: []
    };
   
    ptwData.forEach(row => {
      let pDate = new Date(row[colDate]);
      if (isNaN(pDate)) return;
      pDate.setHours(0, 0, 0, 0);
     
      // STRICT MTD FILTER: Must be this month, and CANNOT be after the selected date
      if (pDate.getFullYear() !== tDateObj.getFullYear() || pDate.getMonth() !== tDateObj.getMonth()) return;
      if (pDate > tDateObj) return;
     
let isToday = (pDate.getTime() === tDateObj.getTime());
     
      let isHot = false, isHgt = false, isCnf = false, isLin = false, isExc = false, isHst = false;
      let rowString = "";


      // If a specific Permit Type column exists, use it. Otherwise, stringify the whole row and search it.
      if (colType > -1) {
        rowString = row[colType] ? row[colType].toString().toLowerCase() : "";
      } else {
        rowString = row.join(" ").toLowerCase();
      }


      if (rowString.includes('hot')) isHot = true;
      if (rowString.includes('height')) isHgt = true;
      if (rowString.includes('confined')) isCnf = true;
      if (rowString.includes('line break')) isLin = true;
      if (rowString.includes('excavation')) isExc = true;
      if (rowString.includes('hoist') || rowString.includes('crane')) isHst = true;
     
      let isGen = (!isHot && !isHgt && !isCnf && !isLin && !isExc && !isHst);
     
// Gather specific types for the description string
      let types = [];
      if (isGen) types.push("General");
      if (isHot) types.push("Hot Work");
      if (isHgt) types.push("Height Work");
      if (isCnf) types.push("Confined Space");
      if (isLin) types.push("Line Breaking");
      if (isExc) types.push("Excavation");
      if (isHst) types.push("Hoisting");
      let typeStr = types.join(", ") || "General";


      if (isGen) { permits.gen.m++; if (isToday) permits.gen.t++; }
      if (isHot) { permits.hot.m++; if (isToday) permits.hot.t++; }
      if (isHgt) { permits.hgt.m++; if (isToday) permits.hgt.t++; }
      if (isCnf) { permits.cnf.m++; if (isToday) permits.cnf.t++; }
      if (isLin) { permits.lin.m++; if (isToday) permits.lin.t++; }
      if (isExc) { permits.exc.m++; if (isToday) permits.exc.t++; }
if (isHst) { permits.hst.m++; if (isToday) permits.hst.t++; }


      // Capture ALL permits for the Month-to-Date table
      let pNo = colPermitNo > -1 ? row[colPermitNo].toString().trim() : "";
      // If Permit No is blank or column doesn't exist, use the Timestamp as the ID
      if (!pNo || pNo === "") pNo = Utilities.formatDate(pDate, Session.getScriptTimeZone(), "MMdd-HHmm");
     
      let ven = colVendor > -1 ? row[colVendor] : "";
      let loc = colLocation > -1 ? row[colLocation] : "";
      let vendorLoc = [ven, loc].filter(Boolean).join(" - ") || "-";
      let desc = colDesc > -1 ? row[colDesc] : "-";
      let pDateFormatted = Utilities.formatDate(pDate, Session.getScriptTimeZone(), "yyyy-MM-dd");


      // We keep the array name 'todayDetails' to avoid breaking the payload, but it now holds MTD data
      permits.todayDetails.push({
         date: pDateFormatted,
         permitNo: pNo,
         type: typeStr,
         vendorLocation: vendorLoc,
         description: desc
      });
    });
   
    return permits;
  } catch (e) {
    Logger.log("PTW Fetch Error: " + e);
    return null;
  }
}


function processDailySubmission(formData) {
  try {
    if (!formData) formData = {};
    const targetDateStr = formData.date || Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
    const ss = SpreadsheetApp.openById(MASTER_DB_ID);
   
    // --- 1. HANDLE PRODUCTION DATA (UPSERT) ---
    const prodSheet = ss.getSheetByName('Production');
    const prodData = prodSheet.getDataRange().getValues();
    if (prodData.length > 0) prodData.shift();


    let existingRowIndex = -1;
    for (let i = 0; i < prodData.length; i++) {
      let rowDate = prodData[i][0];
      let rowDateStr = rowDate instanceof Date ? Utilities.formatDate(rowDate, Session.getScriptTimeZone(), "yyyy-MM-dd") : rowDate;
      if (rowDateStr === targetDateStr) { existingRowIndex = i; break; }
    }


    if (existingRowIndex === -1 && prodData.length > 0) {
      let lastRow = prodData[prodData.length - 1];
      let lastDateObj = new Date(lastRow[0]);
      lastDateObj.setHours(0, 0, 0, 0);
     
      const tDateObj = new Date(targetDateStr);
      tDateObj.setHours(0, 0, 0, 0);
     
      let currDate = new Date(lastDateObj);
      currDate.setDate(currDate.getDate() + 1);
     
      let prevMtd = parseFloat(lastRow[4]) || 0;
      let prevCum = parseFloat(lastRow[7]) || 0;
      let prevTotalMonth = parseFloat(lastRow[9]) || 0;
     
      let workingDays = 0;
      for (let i = 0; i < prodData.length; i++) {
        let rowDate = new Date(prodData[i][0]);
        // Only count working days from the same month as the last row
        if (!isNaN(rowDate) && rowDate.getMonth() === lastDateObj.getMonth() && rowDate.getFullYear() === lastDateObj.getFullYear()) {
          if (parseFloat(prodData[i][3]) > 0) workingDays++;
        }
      }


      let gapRows = [];
      let currentTrackingMonth = lastDateObj.getMonth();
     
      while (currDate < tDateObj) {
        // CRITICAL: If the gap crosses into a new month, reset all MTD/Cum totals to 0!
        if (currDate.getMonth() !== currentTrackingMonth) {
          prevMtd = 0;
          prevCum = 0;
          prevTotalMonth = 0;
          workingDays = 0;
          currentTrackingMonth = currDate.getMonth();
        }
       
        let achieved = workingDays > 0 ? (prevMtd / workingDays) : 0;
        let dStr = Utilities.formatDate(currDate, Session.getScriptTimeZone(), "yyyy-MM-dd");
        gapRows.push([
          dStr, "525 KL", "20 KL", "0.000", prevMtd.toFixed(3), achieved.toFixed(3),
          "0.00", prevCum.toFixed(2), "0.00", prevTotalMonth.toFixed(2), "0", "0", "No Production"
        ]);
        currDate.setDate(currDate.getDate() + 1);
      }
     
      if (gapRows.length > 0) {
        prodSheet.getRange(prodSheet.getLastRow() + 1, 1, gapRows.length, gapRows[0].length).setValues(gapRows);
        prodData.push(...gapRows);
      }
    }


    // DECLARE ONCE FOR BOTH HISTORY FETCH AND CALCULATIONS
    const tDateObjFilter = new Date(targetDateStr);
    tDateObjFilter.setHours(0, 0, 0, 0);
    const targetYear = tDateObjFilter.getFullYear();
    const targetMonth = tDateObjFilter.getMonth();


    let previousMtd = 0;
    let previousCum = 0;
    let workingDaysSoFar = 0;
   
    let loopLimit = existingRowIndex > -1 ? existingRowIndex : prodData.length;
    let lastRowInSameMonth = null;
   
    // CRITICAL: Only fetch historical MTD/Cum data if it belongs to the CURRENT month
    for (let i = 0; i < loopLimit; i++) {
      let rowDate = new Date(prodData[i][0]);
      if (isNaN(rowDate)) continue;
     
      if (rowDate.getFullYear() === targetYear && rowDate.getMonth() === targetMonth) {
        if (parseFloat(prodData[i][3]) > 0) workingDaysSoFar++;
        lastRowInSameMonth = prodData[i];
      }
    }
   
    // If it's the 1st of the month, lastRowInSameMonth remains null, so previousMtd stays 0.
    if (lastRowInSameMonth) {
      previousMtd = parseFloat(lastRowInSameMonth[4]) || 0;
      previousCum = parseFloat(lastRowInSameMonth[7]) || 0;
    }


    const todayProd = parseFloat(formData.todayProduction) || 0;
    const todayDisp = parseFloat(formData.todayDispatched) || 0;
    const pendingOrder = parseFloat(formData.pendingOrder) || 0;
    const manualTotalMonthOrder = parseFloat(formData.totalMonthOrder) || 0;


    const mtdProduction = previousMtd + todayProd;
    if (todayProd > 0) {
      workingDaysSoFar++;
    }
   
    const achievedProdRate = workingDaysSoFar > 0 ? (mtdProduction / workingDaysSoFar) : 0;
    const cumDispatch = previousCum + todayDisp;


    let newRowData = [
      targetDateStr, "525 KL", "20 KL", todayProd.toFixed(3), mtdProduction.toFixed(3),
      achievedProdRate.toFixed(3), todayDisp.toFixed(2), cumDispatch.toFixed(2),
      pendingOrder.toFixed(2), manualTotalMonthOrder.toFixed(2), formData.skus || "0", formData.batches || "0", formData.remarks || "-"
    ];


    if (existingRowIndex > -1) {
      prodSheet.getRange(existingRowIndex + 2, 1, 1, newRowData.length).setValues([newRowData]);
    } else {
      prodSheet.appendRow(newRowData);
    }


    const updatedProdData = prodSheet.getDataRange().getValues();
    if (updatedProdData.length > 0) updatedProdData.shift();
   
    const productionHistory = [];
    updatedProdData.forEach(row => {
      let d = row[0];
      let rowDateObj = (d instanceof Date) ? d : new Date(d);
      rowDateObj.setHours(0, 0, 0, 0);
     
      // STRICT MONTH-WISE FILTER: Only push rows for the current selected month & year
      if (rowDateObj.getFullYear() === targetYear && rowDateObj.getMonth() === targetMonth && rowDateObj <= tDateObjFilter) {
        let dStr = (d instanceof Date) ? Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd") : d;
        productionHistory.push({
          date: dStr, monthlyPlan: row[1], askingRate: row[2], todayProduction: Number(row[3]).toFixed(3),
          mtdProduction: Number(row[4]).toFixed(3), achievedProdRate: Number(row[5]).toFixed(3),
          todayDispatched: Number(row[6]).toFixed(2), cumDispatch: Number(row[7]).toFixed(2),
          pendingOrder: Number(row[8]).toFixed(2), totalMonthOrder: Number(row[9]).toFixed(2), skus: row[10], batches: row[11]
        });
      }
    });


    // --- 2. FETCH LIVE EHS DATA ---
    const safetyData = fetchLiveSafetyData(targetDateStr);
    const ptwData = fetchLivePTWData(targetDateStr);
   
    if (ptwData) {
      formData.l_gen_t = ptwData.gen.t; formData.l_gen_m = ptwData.gen.m;
      formData.l_hot_t = ptwData.hot.t; formData.l_hot_m = ptwData.hot.m;
      formData.l_hgt_t = ptwData.hgt.t; formData.l_hgt_m = ptwData.hgt.m;
      formData.l_cnf_t = ptwData.cnf.t; formData.l_cnf_m = ptwData.cnf.m;
      formData.l_exc_t = ptwData.exc.t; formData.l_exc_m = ptwData.exc.m;
      formData.l_lin_t = ptwData.lin.t; formData.l_lin_m = ptwData.lin.m;
      formData.l_hst_t = ptwData.hst.t; formData.l_hst_m = ptwData.hst.m;
    }


    let ua_today = 0, uc_today = 0;
    safetyData.register.forEach(r => {
      if (r.date === targetDateStr) {
        if (r.type === 'Unsafe Act') ua_today++;
        if (r.type === 'Unsafe Condition') uc_today++;
      }
    });
   
    formData.l_ua_t = ua_today;
    formData.l_uc_t = uc_today;
   
    let ua_total = safetyData.summary.totals.find(t => t.type === 'Unsafe Act');
    let uc_total = safetyData.summary.totals.find(t => t.type === 'Unsafe Condition');
    formData.l_ua_m = ua_total ? ua_total.total : 0;
    formData.l_uc_m = uc_total ? uc_total.total : 0;


    // --- 3. COMPILE PDF ---
    const htmlTemplate = HtmlService.createTemplateFromFile('PdfTemplate');
    htmlTemplate.prodHistory = productionHistory;
    htmlTemplate.ehsData = safetyData;
    htmlTemplate.registerData = safetyData.register;
    htmlTemplate.ptwDetails = ptwData ? ptwData.todayDetails : []; // Pass PTW Details
    htmlTemplate.formData = formData;
    htmlTemplate.remarks = formData.remarks || "-";


    const htmlOutput = htmlTemplate.evaluate().getContent();
    const pdfBlob = Utilities.newBlob(htmlOutput, 'text/html', 'Master_Report.html').getAs('application/pdf');


    return {
      success: true,
      message: "Data logged and live EHS portals synced successfully for " + targetDateStr + ".",
      fileName: `SBL_Master_Report_${targetDateStr}.pdf`,
      fileData: Utilities.base64Encode(pdfBlob.getBytes())
    };
  } catch (error) {
    return { success: false, message: error.toString() };
  }
}
function fetchProductionDataForDate(targetDateStr) {
  try {
    const ss = SpreadsheetApp.openById(MASTER_DB_ID);
    const prodSheet = ss.getSheetByName('Production');
    const prodData = prodSheet.getDataRange().getValues();
    if (prodData.length > 0) prodData.shift();
   
    for (let i = 0; i < prodData.length; i++) {
      let rowDate = prodData[i][0];
      let rowDateStr = rowDate instanceof Date ? Utilities.formatDate(rowDate, Session.getScriptTimeZone(), "yyyy-MM-dd") : rowDate;
      if (rowDateStr === targetDateStr) {
        return {
          found: true,
          todayProduction: prodData[i][3],
          todayDispatched: prodData[i][6],
          pendingOrder: prodData[i][8],
          totalMonthOrder: prodData[i][9],
          skus: prodData[i][10],
          batches: prodData[i][11],
          remarks: prodData[i][12]
        };
      }
    }
    return { found: false };
  } catch (e) {
    return { found: false, error: e.toString() };
  }
}
